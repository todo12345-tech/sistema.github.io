// ========== SISTEMA DE NOTIFICACIONES ==========
class Notificaciones {
    static notificacionesActivas = [];
    static maxNotificaciones = 5;

    static mostrar(mensaje, tipo = 'info', duracion = 5000) {
        let container = document.getElementById('notificationContainer');
        if (!container) {
            container = document.createElement('div');
            container.id = 'notificationContainer';
            container.className = 'notification-container';
            document.body.appendChild(container);
        }

        const notification = document.createElement('div');
        notification.className = `notification ${tipo}`;
        notification.innerHTML = `
            <i class="fas fa-${this.obtenerIcono(tipo)}"></i>
            <span>${mensaje}</span>
            <button class="notification-close">&times;</button>
        `;
        container.appendChild(notification);

        const notificationData = { element: notification, timer: null };
        this.notificacionesActivas.push(notificationData);

        setTimeout(() => notification.classList.add('show'), 10);

        notification.querySelector('.notification-close').addEventListener('click', () => {
            this.cerrar(notificationData);
        });

        if (duracion > 0) {
            notificationData.timer = setTimeout(() => this.cerrar(notificationData), duracion);
        }

        if (this.notificacionesActivas.length > this.maxNotificaciones) {
            const oldNotification = this.notificacionesActivas.shift();
            this.cerrar(oldNotification);
        }

        return notificationData;
    }

    static obtenerIcono(tipo) {
        const iconos = {
            'success': 'check-circle',
            'error': 'exclamation-circle',
            'warning': 'exclamation-triangle',
            'info': 'info-circle'
        };
        return iconos[tipo] || 'info-circle';
    }

    static cerrar(notificationData) {
        if (!notificationData) return;
        const index = this.notificacionesActivas.indexOf(notificationData);
        if (index > -1) {
            this.notificacionesActivas.splice(index, 1);
        }
        if (notificationData.timer) {
            clearTimeout(notificationData.timer);
        }
        notificationData.element.classList.remove('show');
        setTimeout(() => {
            if (notificationData.element.parentNode) {
                notificationData.element.parentNode.removeChild(notificationData.element);
            }
        }, 300);
    }

    static limpiarTodas() {
        while (this.notificacionesActivas.length > 0) {
            this.cerrar(this.notificacionesActivas[0]);
        }
    }
}

// ========== EXTRAER DATOS DEL TEXTO DEL PEDIDO ==========
function extraerDatosDelPedido() {
    const texto = cargarPedidoTextarea.value.trim();

    if (!texto) {
        Notificaciones.mostrar('Por favor pega el texto del pedido primero', 'warning');
        return;
    }

    try {
        let nombre = '';
        let telefono = '';
        let direccion = '';
        let departamento = '';
        let productosTexto = '';

        const nombreMatch = texto.match(/(?:👤\s*)?\*?Nombre\*?:?\s*([^\n]+)/i);
        if (nombreMatch) {
            nombre = nombreMatch[1].trim().replace(/\*/g, '');
        }

        const telefonoMatch = texto.match(/(?:📱\s*)?\*?Tel(?:éfono|efono)?\*?:?\s*(\d+)/i);
        if (telefonoMatch) {
            telefono = telefonoMatch[1].trim();
        }

        const direccionMatch = texto.match(/(?:📍\s*)?\*?Direcci[oó]n\*?:?\s*([^\n]+)/i);
        if (direccionMatch) {
            direccion = direccionMatch[1].trim().replace(/\*/g, '');
        }

        const departamentoMatch = texto.match(/(?:🏢\s*)?\*?Departamento\*?:?\s*([^\n]+)/i);
        if (departamentoMatch) {
            departamento = departamentoMatch[1].trim().replace(/\*/g, '');
        }

        const detalleMatch = texto.match(/(?:📋\s*)?\*?DETALLE DEL PEDIDO\*?:?([\s\S]*?)(?:💰|$)/i);
        if (detalleMatch) {
            productosTexto = detalleMatch[1].trim();
        }

        const lineasProductos = productosTexto.split('\n').filter(linea => linea.includes('➡️'));

        let productosEncontrados = 0;

        lineasProductos.forEach(linea => {
            const lineaLimpia = linea.replace(/➡️\s*/, '').trim();

            let descripcion = '';
            let precio = 0;

            const ultimoGuion = lineaLimpia.lastIndexOf(' - ');

            if (ultimoGuion !== -1) {
                descripcion = lineaLimpia.substring(0, ultimoGuion).trim();
                const precioTexto = lineaLimpia.substring(ultimoGuion + 3).trim();
                const precioMatch = precioTexto.match(/(\d+(?:[.,]\d{1,2})?)/);
                if (precioMatch) {
                    precio = parseFloat(precioMatch[1].replace(',', '.'));
                }
            } else {
                descripcion = lineaLimpia;
            }

            if (descripcion && precio > 0) {
                items.push({
                    id: Date.now() + productosEncontrados,
                    descripcion: descripcion,
                    monto: precio
                });
                productosEncontrados++;
            }
        });

        if (nombre) {
            nombreClienteActual = nombre;
            clienteNombreInput.value = nombre;
        }
        if (direccion) {
            direccionClienteActual = direccion;
            clienteDireccionInput.value = direccion;
        }
        if (departamento) {
            departamentoClienteActual = departamento;
            clienteDepartamentoInput.value = departamento;
        }

        if (productosEncontrados > 0) {
            actualizarListaItems();
            actualizarPrevisualizacion();

            if (!productosContent.classList.contains('expanded')) {
                productosContent.classList.add('expanded');
                productosHeader.querySelector('.collapsible-icon').classList.add('rotated');
            }

            Notificaciones.mostrar(`${productosEncontrados} productos extraídos exitosamente`, 'success');

            let infoCliente = 'Cliente: ';
            if (nombre) infoCliente += nombre;
            if (telefono) infoCliente += ` (Tel: ${telefono})`;
            if (direccion) infoCliente += ` - ${direccion}`;
            if (departamento) infoCliente += `, ${departamento}`;
            Notificaciones.mostrar(infoCliente, 'info', 5000);

            cargarPedidoTextarea.value = '';
        } else {
            Notificaciones.mostrar('No se pudieron extraer productos del texto', 'warning');
        }

    } catch (error) {
        console.error('Error al extraer datos:', error);
        Notificaciones.mostrar('Error al procesar el texto del pedido', 'error');
    }
}

            sucursal: venta.sucursal || '',
            'ubicación': venta.ubicacion || '',
            productos: venta.productos || '',
            montos: venta.montos || '',
            total_venta: venta.total_venta || '0',
            timestamp: venta.timestamp || new Date().toISOString(),
            tipo_sucursal: venta.tipo_sucursal || '',
            usuario: venta.usuario || '',
            observaciones: venta.observaciones || '',
            metodo_pago: venta.metodo_pago || '',
            monto_efectivo: montoEfectivo,
            monto_qr: montoQR,
            apertura_caja: aperturaCaja,
            cierre_caja: cierreCaja,
            dia: venta.dia || '',
            sincronizada: 'TRUE',
            numero_productos: venta.numero_productos || '0',
            direccion: venta.direccion || '',
            departamento: venta.departamento || ''
        };

        Object.keys(datosParaSheetDB).forEach(key => {
            if (datosParaSheetDB[key] === undefined) {
                datosParaSheetDB[key] = '';
            }
        });

        const response = await fetch(url, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ data: datosParaSheetDB })
        });

        if (response.ok) {
            const ventaIndex = ventasDelDia.findIndex(v => v.id === venta.id);
            if (ventaIndex !== -1) {
                ventasDelDia[ventaIndex].sincronizada = true;
                localStorage.setItem('ventasDelDia', JSON.stringify(ventasDelDia));
            }
            Notificaciones.mostrar('Datos enviados exitosamente a SheetDB', 'success', 3000);
            cloudStatus.innerHTML = '<i class="fas fa-circle online"></i> <span>Sincronizado</span>';
            cloudStatus.className = 'right-sidebar-status online';
            return true;
        } else {
            console.error('Error de SheetDB:', await response.text());
            Notificaciones.mostrar('Error al enviar datos a SheetDB', 'error', 3000);
            return false;
        }
    } catch (error) {
        console.error('Error al sincronizar con SheetDB:', error);
        Notificaciones.mostrar('Error de conexión con SheetDB', 'error', 3000);
        return false;
    }
}

async function sincronizarVentasANube() {
    const ventasParaSincronizar = ventasDelDia.filter(v => !v.sincronizada);
    if (ventasParaSincronizar.length === 0) {
        Notificaciones.mostrar('No hay ventas nuevas para sincronizar', 'info');
        return;
    }

    try {
        syncToCloudBtn.disabled = true;
        syncToCloudBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Sincronizando...';
        let ventasEnviadas = 0;
        let errores = 0;

        for (const venta of ventasParaSincronizar) {
            try {
                const exito = await sincronizarVentaIndividualANube(venta);
                if (exito) {
                    ventasEnviadas++;
                } else {
                    errores++;
                }
            } catch (error) {
                console.error(`Error al enviar venta ${venta.id}:`, error);
                errores++;
            }
        }

        if (errores === 0) {
            Notificaciones.mostrar(`${ventasEnviadas} ventas sincronizadas exitosamente`, 'success');
        } else {
            Notificaciones.mostrar(`${ventasEnviadas} ventas sincronizadas, ${errores} errores`, 'warning');
        }
        actualizarContadorVentas();
    } catch (error) {
        console.error('Error al sincronizar:', error);
        Notificaciones.mostrar('Error al sincronizar ventas con la nube: ' + error.message, 'error');
    } finally {
        syncToCloudBtn.disabled = false;
        syncToCloudBtn.innerHTML = '<i class="fas fa-cloud-upload-alt"></i> Subir';
    }
}

async function sincronizarDesdeNube() {
    try {
        syncFromCloudBtn.disabled = true;
        syncFromCloudBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Descargando...';
        const url = obtenerUrlSheetDB(sucursalSeleccionada);
        const response = await fetch(url);

        if (response.ok) {
            const datos = await response.json();
            if (datos.length > 0) {
                crearBackupAutomatico();
                datos.forEach(venta => {
                    venta.sincronizada = true;
                    if (!venta.ubicacion && venta['ubicación']) {
                        venta.ubicacion = venta['ubicación'];
                    }
                });

                const ventasSucursal = datos.filter(v => v.sucursal === sucursalSeleccionada);
                ventasSucursal.forEach(nuevaVenta => {
                    const existe = ventasDelDia.find(v => v.id === nuevaVenta.id);
                    if (!existe) {
                        ventasDelDia.push(nuevaVenta);
                    }
                });

                localStorage.setItem('ventasDelDia', JSON.stringify(ventasDelDia));
                actualizarContadorVentas();
                actualizarEstadisticas();
                actualizarVentasDelDia();
                Notificaciones.mostrar(`Datos descargados exitosamente (${ventasSucursal.length} registros)`, 'success');
                cloudStatus.innerHTML = '<i class="fas fa-circle online"></i> <span>Sincronizado</span>';
                cloudStatus.className = 'right-sidebar-status online';
            } else {
                Notificaciones.mostrar('No hay datos en la hoja de esta sucursal', 'info');
            }
        } else {
            throw new Error('Error al obtener datos de la nube');
        }
    } catch (error) {
        console.error('Error al descargar:', error);
        Notificaciones.mostrar('Error al descargar datos de la nube', 'error');
    } finally {
        syncFromCloudBtn.disabled = false;
        syncFromCloudBtn.innerHTML = '<i class="fas fa-cloud-download-alt"></i> Bajar';
    }
}

async function guardarVentaConMetodoPago() {
    if (guardandoVenta) return;
    guardandoVenta = true;
    confirmMetodoPago.disabled = true;
    confirmMetodoPago.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Guardando...';

    if (metodoPagoSeleccionado === 'Mixto') {
        const montoEfectivo = parseFloat(montoEfectivoMixto.value) || 0;
        const montoQR = parseFloat(montoQRMixto.value) || 0;
        const totalPagado = montoEfectivo + montoQR;
        if (Math.abs(totalPagado - total) > 0.01) {
            Notificaciones.mostrar('La suma de los montos debe ser igual al total de la venta', 'error');
            guardandoVenta = false;
            confirmMetodoPago.disabled = false;
            confirmMetodoPago.innerHTML = '<i class="fas fa-check"></i> Confirmar';
            return;
        }
    }

    try {
        actionGuardarVenta.disabled = true;
        actionGuardarVenta.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Guardando...';
        const ventaGuardada = guardarVentaLocal();
        if (ventaGuardada) {
            Notificaciones.mostrar('Venta guardada correctamente en local', 'success', 3000);
            setTimeout(async () => {
                await sincronizarVentaIndividualANube(ventaGuardada);
            }, 500);
        }
    } catch (error) {
        console.error('Error:', error);
        Notificaciones.mostrar('Error al procesar la venta', 'error');
    } finally {
        guardandoVenta = false;
        actionGuardarVenta.disabled = false;
        actionGuardarVenta.innerHTML = '<i class="fas fa-save"></i> Guardar Venta';
        confirmMetodoPago.disabled = false;
        confirmMetodoPago.innerHTML = '<i class="fas fa-check"></i> Confirmar';
        metodoPagoModal.style.display = 'none';
    }
}

function guardarVentaLocal() {
    const datosSucursal = obtenerDatosSucursal();
    let metodoPagoData = { metodo_pago: metodoPagoSeleccionado };

    if (metodoPagoSeleccionado === 'Mixto') {
        metodoPagoData.monto_efectivo = parseFloat(montoEfectivoMixto.value) || 0;
        metodoPagoData.monto_qr = parseFloat(montoQRMixto.value) || 0;
    } else {
        metodoPagoData.monto_efectivo = '';
        metodoPagoData.monto_qr = '';
    }

    const dias = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
    const fecha = new Date();
    const diaSemana = dias[fecha.getDay()];

    const datosVenta = {
        id: Date.now(),
        nombre_cliente: nombreClienteActual || 'Cliente General',
        direccion: direccionClienteActual || '',
        departamento: departamentoClienteActual || '',
        fecha: new Date().toLocaleDateString('es-ES'),
        hora: new Date().toLocaleTimeString('es-ES'),
        sucursal: datosSucursal.nombre,
        ubicacion: datosSucursal.ubicacion,
        productos: items.map(item => item.descripcion).join(' | '),
        montos: items.map(item => item.monto.toFixed(2)).join(' | '),
        total_venta: total.toFixed(2),
        numero_productos: items.length,
        timestamp: new Date().toISOString(),
        tipo_sucursal: datosSucursal.tipo,
        usuario: currentUser.nombre,
        observaciones: '',
        apertura_caja: '',
        cierre_caja: '',
        sincronizada: false,
        dia: diaSemana,
        ...metodoPagoData
    };

    ventasDelDia.push(datosVenta);
    localStorage.setItem('ventasDelDia', JSON.stringify(ventasDelDia));
    ventasCajaActual.push(datosVenta);
    localStorage.setItem('ventasCajaActual', JSON.stringify(ventasCajaActual));
    crearBackupAutomatico();

    items = [];
    nombreClienteActual = '';
    direccionClienteActual = '';
    departamentoClienteActual = '';
    clienteNombreInput.value = '';
    clienteDireccionInput.value = '';
    clienteDepartamentoInput.value = '';
    actualizarListaItems();
    actualizarPrevisualizacion();
    actualizarContadorVentas();
    actualizarEstadisticas();
    actualizarVentasDelDia();

    return datosVenta;
}

function enviarResumenVentasDiaWhatsApp() {
    const hoy = new Date().toLocaleDateString('es-ES');
    const ventasHoy = ventasDelDia.filter(v => v.fecha === hoy);
    if (ventasHoy.length === 0) {
        Notificaciones.mostrar('No hay ventas hoy para enviar por WhatsApp', 'warning');
        return;
    }

    const totalRecaudado = ventasHoy.reduce((sum, v) => sum + parseFloat(v.total_venta), 0);
    const promedio = totalRecaudado / ventasHoy.length;

    let mensaje = `*RESUMEN DE VENTAS DEL DÍA - TODOENUNO_STORE*%0A%0A`;
    mensaje += `*Fecha:* ${hoy}%0A`;
    mensaje += `*Sucursal:* ${sucursalSeleccionada}%0A`;
    mensaje += `*Total Ventas:* ${ventasHoy.length}%0A`;
    mensaje += `*Total Recaudado:* ${totalRecaudado.toFixed(2)} Bs%0A`;
    mensaje += `*Promedio por Venta:* ${promedio.toFixed(2)} Bs%0A%0A`;
    mensaje += `*DETALLE DE VENTAS:*%0A`;

    ventasHoy.forEach((venta, index) => {
        const productosPreview = venta.productos.split(' | ').slice(0, 2).join(', ');
        const productosExtra = venta.productos.split(' | ').length > 2 ? '...' : '';
        mensaje += `${index + 1}. ${productosPreview}${productosExtra} - ${venta.total_venta} Bs%0A`;
    });

    mensaje += `%0A*MÉTODOS DE PAGO:*%0A`;
    const efectivo = ventasHoy.filter(v => v.metodo_pago === 'Efectivo').reduce((sum, v) => sum + parseFloat(v.total_venta), 0);
    const qr = ventasHoy.filter(v => v.metodo_pago === 'QR').reduce((sum, v) => sum + parseFloat(v.total_venta), 0);
    const mixto = ventasHoy.filter(v => v.metodo_pago === 'Mixto').reduce((sum, v) => sum + parseFloat(v.total_venta), 0);

    mensaje += `Efectivo: ${efectivo.toFixed(2)} Bs%0A`;
    mensaje += `QR: ${qr.toFixed(2)} Bs%0A`;
    mensaje += `Mixto: ${mixto.toFixed(2)} Bs%0A%0A`;
    mensaje += `*Generado por:* ${currentUser.nombre}%0A`;
    mensaje += `*Hora de generación:* ${new Date().toLocaleTimeString('es-ES')}`;

    const whatsappNumber = "75000435";
    const whatsappUrl = `https://wa.me/${whatsappNumber}?text=${mensaje}`;
    window.open(whatsappUrl, '_blank');
}

// abrirCaja(registrarApertura)
//   true  → registra apertura normal (con monto, sincroniza a SheetDB)
//   false → activa el estado "caja abierta" SIN registrar apertura nueva
//           (útil cuando se recargó la página y ya se había abierto caja antes)
function abrirCaja(registrarApertura = true) {
    cajaAbierta = true;

    if (registrarApertura) {
        horaAperturaCaja = new Date().toISOString();
        ventasCajaActual = [];
        localStorage.setItem('cajaAbierta', 'true');
        localStorage.setItem('horaAperturaCaja', horaAperturaCaja);
        localStorage.setItem('ventasCajaActual', JSON.stringify(ventasCajaActual));

        const datosSucursal = obtenerDatosSucursal();
        const dias = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
        const fecha = new Date();
        const diaSemana = dias[fecha.getDay()];

        const registroApertura = {
            id: 'APERTURA',
            fecha: new Date().toLocaleDateString('es-ES'),
            hora: new Date().toLocaleTimeString('es-ES'),
            sucursal: datosSucursal.nombre,
            ubicacion: datosSucursal.ubicacion,
            productos: 'APERTURA DE CAJA',
            montos: '',
            total_venta: '0',
            numero_productos: 0,
            timestamp: new Date().toISOString(),
            tipo_sucursal: datosSucursal.tipo,
            usuario: currentUser.nombre,
            observaciones: '',
            metodo_pago: 'EFECTIVO',
            monto_efectivo: montoApertura.toString(),
            monto_qr: '',
            apertura_caja: montoApertura.toString(),
            cierre_caja: '',
            sincronizada: false,
            dia: diaSemana,
            nombre_cliente: '',
            direccion: '',
            departamento: ''
        };

        ventasDelDia.push(registroApertura);
        localStorage.setItem('ventasDelDia', JSON.stringify(ventasDelDia));
        actualizarEstadoCaja();
        actualizarVentasDelDia();
        Notificaciones.mostrar(`Caja abierta en ${sucursalSeleccionada}`, 'success');

        setTimeout(async () => {
            const exito = await sincronizarVentaIndividualANube(registroApertura);
            if (exito) {
                Notificaciones.mostrar('Apertura de caja sincronizada con SheetDB', 'success');
            }
        }, 1000);
    } else {
        // Continuar sin abrir caja: activamos el estado y permitimos trabajar,
        // pero NO se registra apertura nueva ni se sincroniza a SheetDB.
        if (!horaAperturaCaja) {
            horaAperturaCaja = new Date().toISOString();
        }
        localStorage.setItem('cajaAbierta', 'true');
        localStorage.setItem('horaAperturaCaja', horaAperturaCaja);
        localStorage.setItem('ventasCajaActual', JSON.stringify(ventasCajaActual));

        actualizarEstadoCaja();
        actualizarVentasDelDia();
        Notificaciones.mostrar('Continuando sin abrir caja (no se registra apertura nueva)', 'info', 5000);
    }
}

function cerrarCaja() {
    if (!cajaAbierta) {
        Notificaciones.mostrar('La caja ya está cerrada', 'warning');
        return;
    }

    const datosSucursal = obtenerDatosSucursal();
    const dias = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
    const fecha = new Date();
    const diaSemana = dias[fecha.getDay()];

    const registroCierre = {
        id: 'CIERRE',
        fecha: new Date().toLocaleDateString('es-ES'),
        hora: new Date().toLocaleTimeString('es-ES'),
        sucursal: datosSucursal.nombre,
        ubicacion: datosSucursal.ubicacion,
        productos: 'CIERRE DE CAJA',
        montos: '',
        total_venta: '0',
        numero_productos: 0,
        timestamp: new Date().toISOString(),
        tipo_sucursal: datosSucursal.tipo,
        usuario: currentUser.nombre,
        observaciones: '',
        metodo_pago: 'EFECTIVO',
        monto_efectivo: montoCierre.toString(),
        monto_qr: '',
        apertura_caja: '',
        cierre_caja: 'CIERRE DE CAJA',
        sincronizada: false,
        dia: diaSemana,
        nombre_cliente: '',
        direccion: '',
        departamento: ''
    };

    ventasDelDia.push(registroCierre);
    localStorage.setItem('ventasDelDia', JSON.stringify(ventasDelDia));

    const resumen = {
        efectivo: 0,
        qr: 0,
        tarjeta: 0,
        total: 0,
        numeroVentas: ventasCajaActual.length,
        horaApertura: horaAperturaCaja,
        horaCierre: new Date().toISOString(),
        montoApertura: montoApertura,
        montoCierre: montoCierre
    };

    ventasCajaActual.forEach(venta => {
        const monto = parseFloat(venta.total_venta);
        resumen.total += monto;
        switch(venta.metodo_pago) {
            case 'Efectivo': resumen.efectivo += monto; break;
            case 'QR': resumen.qr += monto; break;
            case 'Tarjeta': resumen.tarjeta += monto; break;
            case 'Mixto':
                resumen.efectivo += venta.monto_efectivo || 0;
                resumen.qr += venta.monto_qr || 0;
                break;
        }
    });

    mostrarResumenCierre(resumen);
    cajaAbierta = false;
    horaAperturaCaja = null;
    ventasCajaActual = [];
    montoApertura = 0;
    montoCierre = 0;
    localStorage.removeItem('cajaAbierta');
    localStorage.removeItem('horaAperturaCaja');
    localStorage.removeItem('ventasCajaActual');
    localStorage.removeItem('montoApertura');
    localStorage.removeItem('montoCierre');
    actualizarEstadoCaja();
    actualizarVentasDelDia();
    Notificaciones.mostrar('Caja cerrada correctamente', 'success');

    setTimeout(async () => {
        const exito = await sincronizarVentaIndividualANube(registroCierre);
        if (exito) {
            Notificaciones.mostrar('Cierre de caja sincronizado con SheetDB', 'success');
        }
    }, 1000);
}

// ========== FUNCIONES RESTANTES ==========
function actualizarPagoMixto() {
    const totalVenta = total;
    const montoEfectivo = parseFloat(montoEfectivoMixto.value) || 0;
    const montoQR = parseFloat(montoQRMixto.value) || 0;
    const totalPagado = montoEfectivo + montoQR;

    document.getElementById('resumenEfectivo').textContent = montoEfectivo.toFixed(2);
    document.getElementById('resumenQR').textContent = montoQR.toFixed(2);
    document.getElementById('totalPagado').textContent = totalPagado.toFixed(2);
    resumenMixto.style.display = 'block';

    const faltante = totalVenta - totalPagado;
    const cambio = totalPagado - totalVenta;

    if (faltante > 0) {
        document.getElementById('mensajeFaltante').style.display = 'block';
        document.getElementById('montoFaltante').textContent = faltante.toFixed(2);
        document.getElementById('mensajeCambio').style.display = 'none';
    } else if (cambio > 0) {
        document.getElementById('mensajeFaltante').style.display = 'none';
        document.getElementById('mensajeCambio').style.display = 'block';
        document.getElementById('montoCambioMixto').textContent = cambio.toFixed(2);
    } else {
        document.getElementById('mensajeFaltante').style.display = 'none';
        document.getElementById('mensajeCambio').style.display = 'none';
    }
}

function calcularCambio() {
    const montoRecibidoValor = parseFloat(montoRecibido.value) || 0;
    if (montoRecibidoValor >= total) {
        const cambio = montoRecibidoValor - total;
        montoCambioElement.textContent = cambio.toFixed(2);
        resultadoCambio.style.display = 'block';
        resultadoCambio.innerHTML = `
            <p><strong>Cambio a Entregar:</strong> <span id="montoCambio">${cambio.toFixed(2)}</span> Bs</p>
            <p style="color: #28a745;"><i class="fas fa-check-circle"></i> Pago completo</p>
        `;
    } else if (montoRecibidoValor > 0 && montoRecibidoValor < total) {
        const faltante = total - montoRecibidoValor;
        resultadoCambio.style.display = 'block';
        resultadoCambio.innerHTML = `
            <p style="color: #dc3545;"><strong>Faltante:</strong> ${faltante.toFixed(2)} Bs</p>
            <p><strong>Monto recibido insuficiente</strong></p>
        `;
    } else {
        resultadoCambio.style.display = 'none';
    }
}

// ========== ACTUALIZAR VENTAS DEL DÍA ==========
function actualizarVentasDelDia() {
    const hoy = new Date().toLocaleDateString('es-ES');
    const todasVentasHoy = ventasDelDia.filter(v => v.fecha === hoy);
    const ventasNormales = todasVentasHoy.filter(v =>
        v.productos !== 'APERTURA DE CAJA' && v.productos !== 'CIERRE DE CAJA'
    );

    const registroApertura = todasVentasHoy.find(v => v.productos === 'APERTURA DE CAJA');
    const registroCierre = todasVentasHoy.find(v => v.productos === 'CIERRE DE CAJA');

    const aperturaMonto = registroApertura ? parseFloat(registroApertura.monto_efectivo) || 0 : 0;
    const cierreMonto = registroCierre ? parseFloat(registroCierre.monto_efectivo) || 0 : 0;

    const totalRecaudado = ventasNormales.reduce((sum, v) => sum + parseFloat(v.total_venta), 0);
    const promedio = ventasNormales.length > 0 ? totalRecaudado / ventasNormales.length : 0;

    const ventasEfectivo = ventasNormales.reduce((sum, v) => {
        if (v.metodo_pago === 'Efectivo') return sum + parseFloat(v.total_venta);
        if (v.metodo_pago === 'Mixto') return sum + (parseFloat(v.monto_efectivo) || 0);
        return sum;
    }, 0);

    const totalEnCaja = aperturaMonto + ventasEfectivo;

    totalVentasDia.textContent = ventasNormales.length;
    totalRecaudadoDia.textContent = totalRecaudado.toFixed(2);
    promedioVentaDia.textContent = promedio.toFixed(2);
    aperturaCajaDia.textContent = aperturaMonto.toFixed(2);
    ventasEfectivoDia.textContent = ventasEfectivo.toFixed(2);
    totalEnCajaDia.textContent = totalEnCaja.toFixed(2);
    cierreEfectivoDia.textContent = cierreMonto.toFixed(2);

    if (todasVentasHoy.length === 0) {
        tablaVentasDia.innerHTML = '';
        noVentasDiaMessage.style.display = 'block';
        return;
    } else {
        noVentasDiaMessage.style.display = 'none';
    }

    let tablaHTML = '';

    todasVentasHoy.forEach(venta => {
        const productosArr = venta.productos.split(' | ');
        const montosArr = venta.montos ? venta.montos.split(' | ') : [];
        const esAperturaCierre = venta.productos === 'APERTURA DE CAJA' || venta.productos === 'CIERRE DE CAJA';

        let productosHTML = '';
        let montosHTML = '';
        let totalVenta = 0;

        if (esAperturaCierre) {
            productosHTML = venta.productos;
            montosHTML = venta.monto_efectivo ? `${venta.monto_efectivo} Bs` : '-';
        } else {
            productosHTML = productosArr.map((p, i) => `<div>${p}</div>`).join('');
            let montosLista = '';
            productosArr.forEach((p, i) => {
                const montoItem = parseFloat(montosArr[i] || 0) || 0;
                totalVenta += montoItem;
                montosLista += `<div>${montoItem.toFixed(2)}</div>`;
            });
            montosHTML = `
                <div class="montos-cell">
                    <div class="montos-lista">${montosLista}</div>
                    <div class="montos-total">${totalVenta.toFixed(2)} Bs</div>
                </div>
            `;
        }

        const tieneObservaciones = venta.observaciones && venta.observaciones.trim() !== '';
        let metodoPagoDisplay = venta.metodo_pago || '';
        if (venta.metodo_pago === 'Mixto') {
            metodoPagoDisplay = `Mixto (Ef: ${venta.monto_efectivo || 0}, QR: ${venta.monto_qr || 0})`;
        }

        const claseFila = esAperturaCierre ? 'fila-apertura-cierre' : '';

        let botonesVentasDia = `
            ${!esAperturaCierre ? `
            <button class="accion-btn observacion-btn" data-index="${ventasDelDia.indexOf(venta)}">
                <i class="fas fa-sticky-note"></i> Obs
            </button>
            <button class="accion-btn imprimir-btn" data-index="${ventasDelDia.indexOf(venta)}">
                <i class="fas fa-print"></i> Imprimir
            </button>
            ` : ''}
            <button class="accion-btn eliminar-btn-tabla" data-index="${ventasDelDia.indexOf(venta)}">
                <i class="fas fa-trash"></i> Eliminar
            </button>
        `;

        const nombreCliente = esAperturaCierre ? venta.usuario : (venta.nombre_cliente || 'Cliente General');
        const direccionCliente = esAperturaCierre ? '-' : (venta.direccion || '-');
        const departamentoCliente = esAperturaCierre ? '-' : (venta.departamento || '-');

        tablaHTML += `
            <tr class="${claseFila}">
                <td>${nombreCliente}</td>
                <td>${direccionCliente}</td>
                <td>${departamentoCliente}</td>
                <td>${venta.hora}</td>
                <td>${productosHTML}</td>
                <td>${montosHTML}</td>
                <td>${metodoPagoDisplay}</td>
                <td>${tieneObservaciones ? venta.observaciones : ''}</td>
                <td>
                    <div class="acciones">${botonesVentasDia}</div>
                </td>
            </tr>
        `;
    });

    tablaVentasDia.innerHTML = tablaHTML;

    document.querySelectorAll('#tablaVentasDia .observacion-btn').forEach(btn => {
        btn.addEventListener('click', function() {
            const index = parseInt(this.getAttribute('data-index'));
            abrirModalObservacion(index);
        });
    });

    document.querySelectorAll('#tablaVentasDia .imprimir-btn').forEach(btn => {
        btn.addEventListener('click', function() {
            const index = parseInt(this.getAttribute('data-index'));
            const venta = ventasDelDia[index];
            imprimirNotaVentaGuardada(venta);
        });
    });

    document.querySelectorAll('#tablaVentasDia .eliminar-btn-tabla').forEach(btn => {
        btn.addEventListener('click', function() {
            const index = parseInt(this.getAttribute('data-index'));
            intentarEliminar(index);
        });
    });
}

// ========== INTENTAR ELIMINAR (con autorización admin) ==========
function intentarEliminar(index) {
    if (currentUser.rol === 'admin') {
        eliminarVenta(index);
    } else {
        ventaAEliminarIndex = index;
        adminUsuarioInput.value = '';
        adminPasswordInput.value = '';
        autorizacionAdminModal.style.display = 'flex';
        setTimeout(() => adminUsuarioInput.focus(), 100);
    }
}

function confirmarAutorizacionEliminar() {
    const usuario = adminUsuarioInput.value.trim().toLowerCase();
    const password = adminPasswordInput.value;

    if (!usuario || !password) {
        Notificaciones.mostrar('Completa usuario y contraseña del administrador', 'warning');
        return;
    }

    if (usuarios[usuario] && usuarios[usuario].password === password && usuarios[usuario].rol === 'admin') {
        const indexAEliminar = ventaAEliminarIndex;
        autorizacionAdminModal.style.display = 'none';
        ventaAEliminarIndex = -1;
        Notificaciones.mostrar(`Autorizado por ${usuarios[usuario].nombre}`, 'success', 3000);
        setTimeout(() => {
            eliminarVenta(indexAEliminar);
        }, 300);
    } else {
        Notificaciones.mostrar('Credenciales de administrador incorrectas', 'error');
    }
}

function cargarVentasDelDia() {
    const ventasGuardadas = localStorage.getItem('ventasDelDia');
    if (ventasGuardadas) {
        ventasDelDia = JSON.parse(ventasGuardadas);
        ventasDelDia.forEach(venta => {
            if (venta.sincronizada === undefined) {
                venta.sincronizada = false;
            }
            if (venta.apertura_caja === undefined) {
                venta.apertura_caja = '';
            }
            if (venta.cierre_caja === undefined) {
                venta.cierre_caja = '';
            }
            if (!venta.ubicacion) {
                const sucursal = sucursales[venta.sucursal];
                if (sucursal) {
                    venta.ubicacion = sucursal.ubicacion;
                }
            }
            if (!venta.dia) {
                const dias = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
                const fecha = new Date(venta.timestamp);
                const diaSemana = dias[fecha.getDay()];
                venta.dia = diaSemana;
            }
            if (!venta.numero_productos) {
                venta.numero_productos = venta.productos ? venta.productos.split(' | ').length : 0;
            }
            if (venta.direccion === undefined) {
                venta.direccion = '';
            }
            if (venta.departamento === undefined) {
                venta.departamento = '';
            }
        });
        localStorage.setItem('ventasDelDia', JSON.stringify(ventasDelDia));
    } else {
        ventasDelDia = [];
        localStorage.setItem('ventasDelDia', JSON.stringify(ventasDelDia));
        crearBackupAutomatico();
    }

    const montoAperturaGuardado = localStorage.getItem('montoApertura');
    if (montoAperturaGuardado) {
        montoApertura = parseFloat(montoAperturaGuardado);
    }
    const montoCierreGuardado = localStorage.getItem('montoCierre');
    if (montoCierreGuardado) {
        montoCierre = parseFloat(montoCierreGuardado);
    }
}

function verificarSesionActiva() {
    const usuarioGuardado = localStorage.getItem('currentUser');
    const cajaAbiertaGuardada = localStorage.getItem('cajaAbierta');
    const horaAperturaGuardada = localStorage.getItem('horaAperturaCaja');
    const ventasCajaGuardadas = localStorage.getItem('ventasCajaActual');
    const sucursalGuardada = localStorage.getItem('sucursalSeleccionada');
    const montoAperturaGuardado = localStorage.getItem('montoApertura');
    const montoCierreGuardado = localStorage.getItem('montoCierre');

    if (usuarioGuardado) {
        currentUser = JSON.parse(usuarioGuardado);
        if (sucursalGuardada) {
            sucursalSeleccionada = sucursalGuardada;
            mostrarPantallaPrincipal();
            if (montoAperturaGuardado) {
                montoApertura = parseFloat(montoAperturaGuardado);
            }
            if (montoCierreGuardado) {
                montoCierre = parseFloat(montoCierreGuardado);
            }
            if (cajaAbiertaGuardada === 'true' && horaAperturaGuardada) {
                cajaAbierta = true;
                horaAperturaCaja = horaAperturaGuardada;
                if (ventasCajaGuardadas) {
                    ventasCajaActual = JSON.parse(ventasCajaGuardadas);
                }
                actualizarEstadoCaja();
            } else {
                cajaAbierta = false;
                horaAperturaCaja = null;
                ventasCajaActual = [];
                actualizarEstadoCaja();
            }
        } else {
            mostrarPantallaSeleccionSucursal();
        }
    }
}

function calcularCierre(efectivoEsperado, qrEsperado, resumen) {
    const efectivoContado = parseFloat(document.getElementById('efectivoContado').value) || 0;
    const qrContado = parseFloat(document.getElementById('qrContado').value) || 0;
    const diferenciaEfectivo = efectivoContado - efectivoEsperado;
    const diferenciaQR = qrContado - qrEsperado;
    const diferenciaTotal = diferenciaEfectivo + diferenciaQR;

    const resultadoEfectivo = document.getElementById('resultadoEfectivo');
    const resultadoQR = document.getElementById('resultadoQR');
    const resultadoTotal = document.getElementById('resultadoTotal');
    const resultadoCierre = document.getElementById('resultadoCierre');

    if (diferenciaEfectivo > 0) {
        resultadoEfectivo.innerHTML = `<strong>Efectivo:</strong> <span class="diferencia-positiva">SOBRANTE ${diferenciaEfectivo.toFixed(2)} Bs</span>`;
    } else if (diferenciaEfectivo < 0) {
        resultadoEfectivo.innerHTML = `<strong>Efectivo:</strong> <span class="diferencia-negativa">FALTANTE ${Math.abs(diferenciaEfectivo).toFixed(2)} Bs</span>`;
    } else {
        resultadoEfectivo.innerHTML = `<strong>Efectivo:</strong> CUADRADO`;
    }

    if (diferenciaQR > 0) {
        resultadoQR.innerHTML = `<strong>QR:</strong> <span class="diferencia-positiva">SOBRANTE ${diferenciaQR.toFixed(2)} Bs</span>`;
    } else if (diferenciaQR < 0) {
        resultadoQR.innerHTML = `<strong>QR:</strong> <span class="diferencia-negativa">FALTANTE ${Math.abs(diferenciaQR).toFixed(2)} Bs</span>`;
    } else {
        resultadoQR.innerHTML = `<strong>QR:</strong> CUADRADO`;
    }

    if (diferenciaTotal > 0) {
        resultadoTotal.innerHTML = `<span class="diferencia-positiva">TOTAL SOBRANTE: ${diferenciaTotal.toFixed(2)} Bs</span>`;
    } else if (diferenciaTotal < 0) {
        resultadoTotal.innerHTML = `<span class="diferencia-negativa">TOTAL FALTANTE: ${Math.abs(diferenciaTotal).toFixed(2)} Bs</span>`;
    } else {
        resultadoTotal.innerHTML = `<span style="color: #4caf50;">TOTAL CUADRADO</span>`;
    }

    resultadoCierre.style.display = 'block';
    document.getElementById('imprimirResumenBtn').style.display = 'inline-block';
}

function realizarBackupCompleto() {
    try {
        const backupData = {
            ventas: ventasDelDia,
            configuracion: {
                usuario: currentUser,
                sucursal: sucursalSeleccionada,
                ultimaActualizacion: new Date().toISOString()
            },
            version: '2.0',
            totalVentas: ventasDelDia.filter(v =>
                v.productos !== 'APERTURA DE CAJA' && v.productos !== 'CIERRE DE CAJA').length,
            totalRecaudado: ventasDelDia.filter(v =>
                v.productos !== 'APERTURA DE CAJA' && v.productos !== 'CIERRE DE CAJA'
            ).reduce((sum, v) => sum + parseFloat(v.total_venta), 0)
        };

        const blob = new Blob([JSON.stringify(backupData, null, 2)], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `backup_todoenuno_${new Date().toISOString().split('T')[0]}.json`;
        a.click();
        URL.revokeObjectURL(url);
        Notificaciones.mostrar('Backup descargado correctamente', 'success');
    } catch (error) {
        console.error('Error al crear backup:', error);
        Notificaciones.mostrar('Error al crear backup', 'error');
    }
}

                    crearBackupAutomatico();
                    ventasDelDia = backupData.ventas;
                    localStorage.setItem('ventasDelDia', JSON.stringify(ventasDelDia));
                    actualizarContadorVentas();
                    actualizarEstadisticas();
                    actualizarVentasDelDia();
                    Notificaciones.mostrar('Backup restaurado correctamente', 'success');
                }
            });
        } catch (error) {
            console.error('Error al leer backup:', error);
            Notificaciones.mostrar('Error: Archivo de backup inválido', 'error');
        }
    };
    reader.readAsText(archivo);
}

function crearBackupAutomatico() {
    try {
        const backup = {
            fecha: new Date().toISOString(),
            ventas: ventasDelDia,
            totalVentas: ventasDelDia.filter(v =>
                v.productos !== 'APERTURA DE CAJA' && v.productos !== 'CIERRE DE CAJA').length,
            totalRecaudado: ventasDelDia.filter(v =>
                v.productos !== 'APERTURA DE CAJA' && v.productos !== 'CIERRE DE CAJA'
            ).reduce((sum, v) => sum + parseFloat(v.total_venta), 0)
        };
        const clave = 'backup_ventas_' + new Date().toISOString().split('T')[0];
        localStorage.setItem(clave, JSON.stringify(backup));
        limpiarBackupsAntiguos();
        console.log('Backup automático creado:', clave);
    } catch (error) {
        console.error('Error creando backup:', error);
    }
}

function limpiarBackupsAntiguos() {
    const sieteDias = 7 * 24 * 60 * 60 * 1000;
    const ahora = Date.now();
    for (let i = 0; i < localStorage.length; i++) {
        const clave = localStorage.key(i);
        if (clave && clave.startsWith('backup_ventas_')) {
            try {
                const fechaBackup = new Date(clave.replace('backup_ventas_', ''));
                if (ahora - fechaBackup.getTime() > sieteDias) {
                    localStorage.removeItem(clave);
                    console.log('Backup antiguo eliminado:', clave);
                }
            } catch (e) {
                console.error('Error procesando backup:', clave, e);
            }
        }
    }
}

function resetSessionTimer() {
    lastActivityTime = Date.now();
    clearTimeout(timeoutTimer);
    timeoutTimer = setTimeout(showSessionWarning, SESSION_TIMEOUT - (WARNING_TIME * 1000));
}

function showSessionWarning() {
    sessionTimeoutOverlay.style.display = 'block';
    sessionTimeoutWarning.style.display = 'block';
    let seconds = WARNING_TIME;
    countdownElement.textContent = seconds;
    countdown = setInterval(() => {
        seconds--;
        countdownElement.textContent = seconds;
        if (seconds <= 0) {
            clearInterval(countdown);
            cerrarSesionAutomatica();
        }
    }, 1000);
}

function hideSessionWarning() {
    sessionTimeoutOverlay.style.display = 'none';
    sessionTimeoutWarning.style.display = 'none';
    clearInterval(countdown);
    resetSessionTimer();
}

function cerrarSesionAutomatica() {
    Notificaciones.mostrar('Sesión cerrada por inactividad', 'warning');
    if (cajaAbierta) {
        cerrarCaja();
    }
    localStorage.removeItem('currentUser');
    localStorage.removeItem('sucursalSeleccionada');
    localStorage.removeItem('montoApertura');
    localStorage.removeItem('montoCierre');
    localStorage.removeItem('cajaAbierta');
    localStorage.removeItem('horaAperturaCaja');
    localStorage.removeItem('ventasCajaActual');
    currentUser = null;
    sucursalSeleccionada = null;
    mainScreen.style.display = 'none';
    loginScreen.style.display = 'block';
    usernameInput.value = '';
    passwordInput.value = '';
}

    usuarioActualNombre.textContent = currentUser.nombre + ' (' + currentUser.rol + ')';
    nuevoUsuarioInput.value = '';
    nuevaPasswordInput.value = '';

function cambiarUsuario() {
    const nuevoUsuario = nuevoUsuarioInput.value.trim().toLowerCase();
    const nuevaPassword = nuevaPasswordInput.value;

    if (!nuevoUsuario || !nuevaPassword) {
        Notificaciones.mostrar('Completa usuario y contraseña', 'warning');
        return;
    }

    if (usuarios[nuevoUsuario] && usuarios[nuevoUsuario].password === nuevaPassword) {
        const cajaAbiertaGuardada = cajaAbierta;
        const horaAperturaGuardada = horaAperturaCaja;
        const ventasCajaGuardadas = [...ventasCajaActual];
        const montoAperturaGuardado = montoApertura;
        const montoCierreGuardado = montoCierre;

        currentUser = {
            username: nuevoUsuario,
            nombre: usuarios[nuevoUsuario].nombre,
            rol: usuarios[nuevoUsuario].rol
        };

        cajaAbierta = cajaAbiertaGuardada;
        horaAperturaCaja = horaAperturaGuardada;
        ventasCajaActual = ventasCajaGuardadas;
        montoApertura = montoAperturaGuardado;
        montoCierre = montoCierreGuardado;

        localStorage.setItem('currentUser', JSON.stringify(currentUser));
        localStorage.setItem('cajaAbierta', cajaAbierta.toString());
        localStorage.setItem('horaAperturaCaja', horaAperturaCaja || '');
        localStorage.setItem('ventasCajaActual', JSON.stringify(ventasCajaActual));
        localStorage.setItem('montoApertura', montoApertura.toString());
        localStorage.setItem('montoCierre', montoCierre.toString());

        currentUserName.textContent = currentUser.nombre;
        currentUserRole.textContent = `(${currentUser.rol})`;

        if (currentUser.rol === 'admin') {
            document.querySelectorAll('.permiso-admin').forEach(el => {
                el.style.display = 'block';
            });
            document.querySelectorAll('.permiso-vendedor').forEach(el => {
                el.style.display = 'none';
            });
        } else {
            document.querySelectorAll('.permiso-admin').forEach(el => {
                el.style.display = 'none';
            });
        }

        actualizarEstadoCaja();
        actualizarVentasDelDia();

        cambiarUsuarioModal.style.display = 'none';
        Notificaciones.mostrar(`Usuario cambiado a: ${currentUser.nombre}`, 'success');

        resetSessionTimer();
    } else {
        Notificaciones.mostrar('Usuario o contraseña incorrectos', 'error');
    }
}
